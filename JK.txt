#############################################################################################################################5
KL transform
clear;  
clc;  

X = [4,3,5,6; 4,2,7,7; 5,5,6,7];  
[m,n] = size(X);  

A = zeros(m,1);  
E = zeros(m,m);  

for i = 1:n  
    A = A + X(:,i);  
    E = E + X(:,i)*X(:,i)';  
end  

mx = A / n;  
E = E / n;  
C = E - mx*mx';  

[V,D] = spec(C);  
d = diag(D);  
[d_sorted, idx] = gsort(d);  

for j = 1:length(d_sorted)  
    T(:,j) = V(:,idx(j));  
end  

T = T';  

disp("Eigen Values are U =")  
disp(d_sorted)  

disp("The eigen vector matrix T =")  
disp(T)  

disp("The KL transform basis is =")  
disp(T)  

for i = 1:n  
    Y(:,i) = T * X(:,i);  
end  

disp("KL transformation of the input matrix Y =")  
disp(Y)  

for i = 1:n  
    X_reconstructed(:,i) = T' * Y(:,i);  
end  

disp("Reconstructed matrix of the given sample matrix X =")  
disp(X_reconstructed)

Block Truncation Coding BTC (Output in the form of Matrix).
Code:
close;
clear;
clc;
x=[65,75,80,70;72,75,82,68;84,72,62,65;66,68,72,80];
disp(x,"Original Block is x = " );
[ m1 n1 ] = size(x);
blk = input("Enter the block size: ");
for i = 1:blk:m1
 for j = 1:blk:n1
 y = x(i:i+(blk-1),j:j+(blk -1));
 m = mean(mean(y));
 disp(m,"mean value is m = ");
 sig = stdev(y);
 disp(sig,"Standard deviation of the block is = ");
 b = y>m;
 disp(b,"Binary allocation matrix is B= ");
 K = sum(sum(b));
 disp(K,"number of ones = ");
 if(K~=blk^2)&( K~=0)
 ml = m-sig*sqrt(K/((blk^2)-K));
 disp(ml,"The value of a = ");
 mu = m+sig*sqrt(((blk^2)-K)/K);
 disp(mu,"The value of b = ");
 x(i:i+(blk-1),j:j+(blk-1))=b*mu+(1-b)*ml;
 end
 end
end
disp(round(x),"Reconstructed Block is x = " );

2.Image Multiplication & Division 
Code:
clc();
clear();
close();
// Load images
img1 = imread('C:\Users\Admin\Downloads\Doremon.jpg');
img2 = imread('C:\Users\Admin\Downloads\newdoremon.jpg');
img1 = rgb2gray(img1);
img2 = rgb2gray(img2);
// Convert to binary (thresholding)
img1 = img1 > 128;
img2 = img2 > 128;
// Resize second image if sizes are different
if size(img1, 'r') ~= size(img2, 'r') | size(img1, 'c') ~= size(img2, 'c') then
    img2 = imresize(img2, size(img1, 'r'), size(img1, 'c'));
end
// Perform Multiplication
multiplied_img = img1 .* img2;
// Perform Division (avoid division by zero)
img2(img2 == 0) = 1;
divided_img = img1 ./ img2;
// Display images
figure();
subplot(1, 4, 1);
imshow(img1);
title("Image 1");
subplot(1, 4, 2);
imshow(img2);
title("Image 2");
subplot(1, 4, 3);
imshow(multiplied_img);
title("Image Multiplication Result");
subplot(1, 4, 4);
imshow(divided_img);
title("Image Division Result");
